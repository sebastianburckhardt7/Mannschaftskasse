        async function supabaseCall(table, method = 'GET', body = null) {

            const headers = {

                'apikey': SUPABASE_KEY,

                'Authorization': `Bearer ${SUPABASE_KEY}`,

                'Content-Type': 'application/json'

            const url = `${SUPABASE_URL}/rest/v1/${table}`;

            const options = { method, headers };

            if (body) options.body = JSON.stringify(body);

            try {

                const response = await fetch(url, options);

                if (!response.ok) throw new Error(`Error: ${response.status}`);

                return await response.json();

            } catch (error) {

                showMessage('Verbindungsfehler');

                return [];

            }

        }

        async function addStrafe() {

        async function addZahlung() {

            const spieler_id = parseInt(document.getElementById('zahlungMitglied').value);

            const betrag = parseFloat(document.getElementById('zahlungBetrag').value);

            const datum = document.getElementById('zahlungDatum').value;

            if (!spieler_id || !betrag || !datum) { showMessage('Alle Felder erforderlich'); return; }

            const result = await supabaseCall('zahlungen', 'POST', { spieler_id, betrag, datum });

            if (result && !result.code) {

                const spielerStrafen = data.strafen.filter(s => s.spieler_id === spieler_id && !s.bezahlt);

                let remaining = betrag;

                        await supabaseCall(`strafen?id=eq.${strafe.id}`, 'PATCH', { bezahlt: true });

                        remaining -= strafe.betrag;

                    }

                }

                document.getElementById('zahlungBetrag').value = '';

                showMessage('Zahlung erfasst');

        }


        async function deleteZahlung(id) {

            if (confirm('Zahlung wirklich löschen?')) {

                await supabaseCall(`zahlungen?id=eq.${id}`, 'DELETE');

                loadData();

            }

        }


        function updateUI() {

            updateDashboard();

            updateMitgliederListe();

            updateStrafenListe();

            updateSelects();

            updateZahlungenListe();

        }


        function updateDashboard() {

            const zahlungen = data.zahlungen.reduce((sum, z) => sum + z.betrag, 0);

            const offen = data.strafen.filter(s => !s.bezahlt).reduce((sum, s) => sum + s.betrag, 0);

            document.getElementById('kassenstand').textContent = '€' + (zahlungen - offen).toFixed(2);

            document.getElementById('offeneBetraege').textContent = '€' + offen.toFixed(2);

            document.getElementById('mitgliederAnzahl').textContent = data.mitglieder.length;

            

            let offeneMitgliederIds = [...new Set(data.strafen.filter(s => !s.bezahlt).map(s => s.spieler_id))];

            let html = '';

            offeneMitgliederIds.forEach(mId => {

                const mitglied = data.mitglieder.find(m => m.id === mId);

                const mitgliedOffen = data.strafen.filter(s => s.spieler_id === mId && !s.bezahlt).reduce((sum, s) => sum + s.betrag, 0);

                if (mitglied) html += `<div class="list-item open"><div><div class="list-item-title">${mitglied.name}</div></div><div class="list-item-amount">€${mitgliedOffen.toFixed(2)}</div></div>`;

            });

            document.getElementById('offeneMitglieder').innerHTML = html || '<p style="color: #666;">Alle bezahlt!</p>';

            

            let strafenCount = {};

            data.strafen.forEach(s => {

                const r = data.strafenkatalog.find(r => r.id === s.strafe_id);

                if (r) strafenCount[r.id] = (strafenCount[r.id] || 0) + 1;

            });

            html = '';

            Object.entries(strafenCount).sort((a, b) => b[1] - a[1]).slice(0, 5).forEach(([id, count]) => {

                const r = data.strafenkatalog.find(r => r.id == id);

                html += `<div style="padding: 8px 0; border-bottom: 1px solid #eee; display: flex; justify-content: space-between; font-size: 12px;"><span>${r?.name}</span><strong>${count}x</strong></div>`;

            });

            document.getElementById('topStrafenAnzahl').innerHTML = html || '<p style="color: #666;">Keine</p>';

        }


        function updateMitgliederListe() {

            let html = '';

            data.mitglieder.forEach(m => {

                html += `<div class="list-item"><div><div class="list-item-title">${m.name}</div></div><button class="btn btn-danger" onclick="deleteMitglied(${m.id})">Löschen</button></div>`;

            });

            document.getElementById('mitgliederListe').innerHTML = html || '<p style="color: #666;">Keine</p>';

        }


        function updateStrafenListe() {

            let html = '';

            data.strafenkatalog.forEach(s => {

                const count = data.strafen.filter(st => st.strafe_id === s.id).length;

                html += `<div class="list-item"><div><div class="list-item-title">${s.name}</div><div class="list-item-subtitle">${count}x</div></div><div style="display: flex; gap: 8px;"><span class="list-item-amount">€${s.betrag.toFixed(2)}</span><button class="btn btn-danger" onclick="deleteStrafenregel(${s.id})">Löschen</button></div></div>`;

            });

            document.getElementById('strafenListe').innerHTML = html || '<p style="color: #666;">Keine</p>';

        }


        function updateSelects() {

            let mHtml = '<option value="">-- Wählen --</option>';

            data.mitglieder.forEach(m => { mHtml += `<option value="${m.id}">${m.name}</option>`; });

            document.getElementById('strafeMitglied').innerHTML = mHtml;

            document.getElementById('zahlungMitglied').innerHTML = mHtml;

            

            let sHtml = '<option value="">-- Wählen --</option>';

            data.strafenkatalog.forEach(s => { sHtml += `<option value="${s.id}">${s.name} (€${s.betrag.toFixed(2)})</option>`; });

            document.getElementById('strafeTyp').innerHTML = sHtml;

        }


        function updateZahlungenListe() {

            let html = '';

            data.zahlungen.forEach(z => {

                const m = data.mitglieder.find(mi => mi.id === z.spieler_id);

                html += `<div class="list-item paid"><div><div class="list-item-title">${m?.name}</div><div class="list-item-subtitle">${z.datum}</div></div><div style="display: flex; gap: 8px;"><span class="list-item-amount">€${z.betrag.toFixed(2)}</span><button class="btn btn-danger" onclick="deleteZahlung(${z.id})">Löschen</button></div></div>`;

            });

            document.getElementById('zahlungenListe').innerHTML = html || '<p style="color: #666;">Keine</p>';

        }


        document.getElementById('strafeDatum').valueAsDate = new Date();

        document.getElementById('zahlungDatum').valueAsDate = new Date();

        setupNav();

        loadData();

        setInterval(loadData, 3000);
